def binary_search(arr, target):
    """Return the index of target in the sorted list arr, or -1 if absent."""

def flatten(nested):
    """Return a flat list containing every element of a list of lists."""

def count_words(text):
    """Return a dict mapping each whitespace-separated word to its count."""

def reverse_string(s):
    """Return the string s with its characters in reverse order."""

def is_palindrome(s):
    """Return True if s reads the same forwards and backwards, ignoring case."""

def fibonacci(n):
    """Return the nth Fibonacci number, with fibonacci(0) == 0 and fibonacci(1) == 1."""

def chunk_list(items, size):
    """Split items into consecutive lists of length size; the last may be shorter."""

def merge_dicts(a, b):
    """Return a new dict with the keys of a and b, where b wins on conflicts."""

def remove_duplicates(items):
    """Return a list with duplicates removed, preserving first-seen order."""

def safe_divide(numerator, denominator, default=0.0):
    """Return numerator / denominator, or default when denominator is zero."""

def read_json_file(path):
    """Load and return the JSON object stored in the file at path."""

def write_lines(path, lines):
    """Write each string in lines to path, one per line."""

def find_max(numbers):
    """Return the largest value in numbers, or None if numbers is empty."""

def average(numbers):
    """Return the arithmetic mean of numbers, or 0.0 if numbers is empty."""

def celsius_to_fahrenheit(celsius):
    """Convert a temperature in Celsius to Fahrenheit and return it."""

def retry(func, attempts=3):
    """Call func() up to attempts times, returning its first successful result.
    Re-raise the last exception if every attempt fails.
    """

def memoize(func):
    """Decorator caching func's return values keyed by its positional arguments."""

def group_by_length(words):
    """Return a dict mapping each word length to the list of words of that length."""

def validate_email(address):
    """Return True if address contains exactly one '@' with text on both sides."""

def parse_key_value(line):
    """Split a 'key=value' string and return the pair (key, value) as strings."""
